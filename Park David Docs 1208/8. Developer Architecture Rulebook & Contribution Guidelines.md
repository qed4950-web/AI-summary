## 2️⃣ `core/data_pipeline/scanner.py` — 스켈레톤 코드

```python
"""
core.data_pipeline.scanner

Smart Folder + Policy 기반 파일 스캐너.

역할:
- SmartFolderContext가 정의한 root / policy 안에서만 파일을 탐색한다.
- allowed / denied 여부를 평가해서 메타데이터를 생성한다.
- 해시/수정시간 기반으로 증분 스캔을 지원할 수 있도록 hook을 제공한다.
- 최종 결과는 CSV(or DataFrame) 형태로 반환하거나 파일로 저장한다.

주의:
- Smart Folder 범위 밖 경로는 절대 접근하지 않는다.
- deny_types, max_file_size_mb 등 정책을 반드시 적용한다.
- 실제 state(이전 스캔 결과) 관리는 state_manager에 위임한다.
"""

from __future__ import annotations

import csv
import hashlib
import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Iterable, List, Optional, Dict, Any

logger = logging.getLogger(__name__)


# === Smart Folder / State 타입 정의 (실제 구현은 별도 모듈에서 제공) ======================

class SmartFolderContext:
    """
    Smart Folder 컨텍스트의 최소 인터페이스 스텁.

    실제 구현(예: core.smartfolder.context.SmartFolderContext)은
    아래 프로퍼티/메서드를 반드시 제공해야 한다.
    """

    root: Path
    recursive: bool
    allowed_types: List[str]        # 예: [".md", ".txt", ".pdf"]
    deny_types: List[str]           # 예: [".exe", ".dll"]
    max_file_size_mb: Optional[int] # None이면 제한 없음

    def is_path_in_scope(self, path: Path) -> bool:
        """해당 경로가 Smart Folder scope 안에 있는지 검사."""
        raise NotImplementedError

    def describe(self) -> str:
        """로깅/디버깅용 설명 문자열."""
        raise NotImplementedError


class ScanStateManager:
    """
    이전 스캔 상태를 관리하는 최소 인터페이스 스텁.

    실제 구현(예: core.data_pipeline.state_manager.ScanStateManager)은
    해시/mtime 기반 증분 스캔을 제공해야 한다.
    """

    def load(self) -> Dict[str, Any]:
        raise NotImplementedError

    def save(self, state: Dict[str, Any]) -> None:
        raise NotImplementedError


# === 스캔 결과 모델 =======================================================================

@dataclass
class ScannedFile:
    """스캔된 개별 파일의 메타데이터."""
    path: Path
    size_bytes: int
    modified_ts: float
    content_hash: Optional[str]
    allowed: bool
    deny_reason: Optional[str] = None

    def to_row(self) -> Dict[str, Any]:
        """CSV/Parquet 저장을 위한 dict 형태로 변환."""
        return {
            "path": str(self.path),
            "size_bytes": self.size_bytes,
            "modified_ts": self.modified_ts,
            "content_hash": self.content_hash or "",
            "allowed": int(self.allowed),
            "deny_reason": self.deny_reason or "",
        }


@dataclass
class ScannerConfig:
    """
    스캐너 동작 옵션.

    SmartFolderContext에서 대부분 유도되지만, 스캐너 자체의 부가 옵션을 넣는다.
    """
    include_hidden: bool = False
    follow_symlinks: bool = False
    calculate_hash: bool = True     # False면 hash 계산 생략 (속도 우선)
    hash_algo: str = "sha256"
    # 추가 옵션 여지를 위해 확장용 dict
    extra: Optional[Dict[str, Any]] = None


# === 메인 스캐너 구현 ======================================================================

class FileScanner:
    """
    Smart Folder + Policy 기반 파일 스캐너.

    주요 책임:
    - SmartFolderContext에 정의된 root 아래만 순회
    - allowed_types / deny_types / max_file_size_mb 정책 적용
    - (선택적으로) content hash 계산
    - 이전 상태(ScanStateManager)와 비교해 증분 스캔도 가능하도록 hook 제공
    """

    def __init__(
        self,
        smartfolder: SmartFolderContext,
        state_manager: Optional[ScanStateManager] = None,
        config: Optional[ScannerConfig] = None,
    ) -> None:
        self.smartfolder = smartfolder
        self.state_manager = state_manager
        self.config = config or ScannerConfig()

    # --- Public API ------------------------------------------------------------------

    def scan(self) -> List[ScannedFile]:
        """
        Smart Folder 범위 안에서 전체 스캔을 수행한다.

        반환:
            ScannedFile 리스트 (allowed/denied 모두 포함)
        """
        logger.info(
            "Starting scan. root=%s, recursive=%s, allowed_types=%s, deny_types=%s",
            self.smartfolder.root,
            getattr(self.smartfolder, "recursive", True),
            getattr(self.smartfolder, "allowed_types", None),
            getattr(self.smartfolder, "deny_types", None),
        )

        previous_state: Dict[str, Any] = {}
        if self.state_manager is not None:
            try:
                previous_state = self.state_manager.load() or {}
                logger.debug(
                    "Loaded previous scan state: %d entries",
                    len(previous_state.get("files", {})),
                )
            except Exception as exc:  # noqa: BLE001
                logger.warning("Failed to load previous scan state: %s", exc)

        results: List[ScannedFile] = []

        for path in self._iter_candidate_files():
            scanned = self._scan_single_file(path, previous_state=previous_state)
            if scanned is None:
                # 스킵된 파일 (범위 밖, 정책 위반 등)
                continue
            results.append(scanned)

        # 새로운 상태 저장 (증분 스캔 지원)
        if self.state_manager is not None:
            new_state = self._build_state_from_results(results)
            try:
                self.state_manager.save(new_state)
                logger.debug(
                    "Saved scan state: %d files",
                    len(new_state.get("files", {})),
                )
            except Exception as exc:  # noqa: BLE001
                logger.warning("Failed to save scan state: %s", exc)

        logger.info("Scan finished. total_files=%d", len(results))
        return results

    def write_csv(self, out_path: Path, scanned_files: Iterable[ScannedFile]) -> None:
        """
        스캔 결과를 CSV 파일로 저장한다.

        컬럼:
            path, size_bytes, modified_ts, content_hash, allowed, deny_reason
        """
        out_path.parent.mkdir(parents=True, exist_ok=True)

        rows = [f.to_row() for f in scanned_files]
        fieldnames = list(rows[0].keys()) if rows else [
            "path",
            "size_bytes",
            "modified_ts",
            "content_hash",
            "allowed",
            "deny_reason",
        ]

        logger.info("Writing scan CSV: %s (rows=%d)", out_path, len(rows))

        with out_path.open("w", newline="", encoding="utf-8") as fp:
            writer = csv.DictWriter(fp, fieldnames=fieldnames)
            writer.writeheader()
            for row in rows:
                writer.writerow(row)

    # --- Internal helpers -------------------------------------------------------------

    def _iter_candidate_files(self) -> Iterable[Path]:
        """Smart Folder root 아래의 후보 파일들을 순회한다."""
        root = self.smartfolder.root

        if not root.exists():
            logger.warning("SmartFolder root does not exist: %s", root)
            return []

        if not root.is_dir():
            logger.warning("SmartFolder root is not a directory: %s", root)
            return []

        recursive = getattr(self.smartfolder, "recursive", True)

        if recursive:
            walker = root.rglob("*")
        else:
            walker = root.glob("*")

        for path in walker:
            if not path.is_file():
                continue
            if not self.smartfolder.is_path_in_scope(path):
                # Smart Folder 범위 밖 → 무시
                logger.debug("Skipping out-of-scope file: %s", path)
                continue
            if not self.config.include_hidden and self._is_hidden(path):
                logger.debug("Skipping hidden file: %s", path)
                continue
            yield path

    def _scan_single_file(
        self,
        path: Path,
        previous_state: Optional[Dict[str, Any]] = None,
    ) -> Optional[ScannedFile]:
        """
        단일 파일에 대해 정책 검사 + 메타데이터 계산.

        previous_state는 증분 스캔 구현용 hook이다.
        """
        st = path.stat()
        size_bytes = st.st_size
        modified_ts = st.st_mtime

        # 정책 검사
        allowed, deny_reason = self._check_policy(path, size_bytes=size_bytes)

        # 허용되지 않은 파일도 로그/CSV에는 남기지만, extractor 이후 단계에서는 제외된다.
        content_hash: Optional[str] = None
        if self.config.calculate_hash and allowed:
            content_hash = self._compute_hash(path)

        scanned = ScannedFile(
            path=path,
            size_bytes=size_bytes,
            modified_ts=modified_ts,
            content_hash=content_hash,
            allowed=allowed,
            deny_reason=deny_reason,
        )

        logger.debug(
            "Scanned file: path=%s size=%d allowed=%s reason=%s hash=%s",
            path,
            size_bytes,
            allowed,
            deny_reason,
            content_hash,
        )
        return scanned

    def _check_policy(self, path: Path, size_bytes: int) -> tuple[bool, Optional[str]]:
        """
        Smart Folder 정책을 기준으로 파일 허용 여부를 판정한다.

        정책 우선순위:
        1) scope 검사 (is_path_in_scope)
        2) deny_types
        3) allowed_types
        4) max_file_size_mb
        """
        # 1) scope
        if not self.smartfolder.is_path_in_scope(path):
            return False, "out_of_scope"

        suffix = path.suffix.lower()

        # 2) deny_types
        deny_types = [ext.lower() for ext in getattr(self.smartfolder, "deny_types", [])]
        if suffix in deny_types:
            return False, "deny_types"

        # 3) allowed_types
        allowed_types = [ext.lower() for ext in getattr(self.smartfolder, "allowed_types", [])]
        if allowed_types and suffix not in allowed_types:
            return False, "not_in_allowed_types"

        # 4) max_file_size_mb
        max_file_size_mb = getattr(self.smartfolder, "max_file_size_mb", None)
        if max_file_size_mb is not None:
            max_bytes = max_file_size_mb * 1024 * 1024
            if size_bytes > max_bytes:
                return False, "file_too_large"

        return True, None

    def _is_hidden(self, path: Path) -> bool:
        """간단한 hidden 파일 판단(Unix 스타일 .prefix)."""
        return path.name.startswith(".")

    def _compute_hash(self, path: Path) -> str:
        """
        파일 내용을 읽어 sha256(기본) 해시를 계산한다.

        성능이 중요할 경우 chunk 크기를 조정하거나,
        일부 prefix만 사용하는 최적화 여지를 남겨둔다.
        """
        algo = self.config.hash_algo.lower()
        try:
            h = hashlib.new(algo)
        except Exception as exc:  # noqa: BLE001
            logger.warning("Unsupported hash algo '%s', fallback to sha256 (%s)", algo, exc)
            h = hashlib.sha256()

        with path.open("rb") as fp:
            for chunk in iter(lambda: fp.read(1024 * 1024), b""):
                h.update(chunk)
        return h.hexdigest()

    def _build_state_from_results(self, results: List[ScannedFile]) -> Dict[str, Any]:
        """
        ScanStateManager에 저장할 상태를 구성한다.

        최소 구성:
        {
          "files": {
             "<path>": {
               "modified_ts": ...,
               "content_hash": ...
             },
             ...
          }
        }
        """
        files_state: Dict[str, Dict[str, Any]] = {}
        for item in results:
            if not item.allowed:
                # 증분 스캔은 allowed 파일 위주로 관리해도 충분하다.
                continue
            files_state[str(item.path)] = {
                "modified_ts": item.modified_ts,
                "content_hash": item.content_hash,
                "size_bytes": item.size_bytes,
            }

        return {"files": files_state}
```

## 2️⃣ `core/data_pipeline/extractor.py` 전체 스켈레톤

```python
"""
core.data_pipeline.extractor

역할:
- scanner 단계에서 만들어진 ScannedFile 목록 또는 CSV를 입력으로 받아,
  허용된(allowed=1) 파일들의 텍스트/메타데이터를 추출한다.
- 결과를 in-memory DataFrame 또는 Parquet/CSV 파일로 저장해
  embedder/trainer/retriever 단계의 공통 코퍼스(corpus)로 사용한다.

설계 포인트:
- 파일 확장자별 extractor registry를 두어, PDF/Office/Markdown 등
  포맷별 파서 구현을 유연하게 확장할 수 있게 한다.
- Smart Folder 정책은 scanner에서 이미 한 번 필터링했지만,
  extractor 단계에서도 최소한의 방어 코드(allowed 플래그, scope)로 이중 방어를 유지한다.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Callable, Dict, Iterable, List, Optional

import pandas as pd

from .scanner import ScannedFile, SmartFolderContext

logger = logging.getLogger(__name__)


# === 데이터 모델 ==========================================================================

@dataclass
class DocumentRecord:
    """
    코퍼스에 들어가는 한 문서 단위 레코드.

    필드 예시:
    - doc_id: 내부 유니크 ID (path 기반 또는 해시 기반)
    - path: 원본 파일 경로
    - title: 파일명 또는 추출된 제목
    - text: 전체 텍스트(후속 단계에서 chunking 대상)
    - metadata: 태그, 폴더, 파일 타입 등의 부가 정보
    """
    doc_id: str
    path: Path
    text: str
    title: Optional[str] = None
    file_type: Optional[str] = None
    folder: Optional[str] = None
    metadata: Optional[Dict[str, Any]] = None

    def to_row(self) -> Dict[str, Any]:
        meta = self.metadata or {}
        row = {
            "doc_id": self.doc_id,
            "path": str(self.path),
            "text": self.text,
            "title": self.title or "",
            "file_type": self.file_type or "",
            "folder": self.folder or "",
        }
        # metadata는 평탄화하거나 별도 컬럼에 JSON string으로 저장하는 방식 중 택1.
        # 여기선 단순히 json-like dict를 하나의 컬럼으로 둔다.
        row["metadata"] = meta
        return row


@dataclass
class ExtractorConfig:
    """
    Extractor 동작 옵션.

    - max_text_length: 너무 긴 문서는 잘라낼 상한.
    - encoding_fallbacks: 텍스트 파일에 대해 시도할 인코딩 목록.
    """
    max_text_length: Optional[int] = None
    encoding_fallbacks: Optional[List[str]] = None
    # 후속 옵션 확장을 위한 extra
    extra: Optional[Dict[str, Any]] = None


# === Extractor Registry 인터페이스 ========================================================

TextExtractorFunc = Callable[[Path], str]


class ExtractorRegistry:
    """
    파일 확장자별 텍스트 추출 함수를 관리하는 레지스트리.

    예:
        registry.register(".txt", extract_plain_text)
        registry.register(".md", extract_markdown)
        registry.register(".pdf", extract_pdf)

    등록되지 않은 확장자는 fallback extractor를 사용한다.
    """

    def __init__(self, fallback_extractor: Optional[TextExtractorFunc] = None) -> None:
        self._by_suffix: Dict[str, TextExtractorFunc] = {}
        self._fallback = fallback_extractor or self._default_fallback

    def register(self, suffix: str, extractor: TextExtractorFunc) -> None:
        key = suffix.lower()
        self._by_suffix[key] = extractor

    def get(self, suffix: str) -> TextExtractorFunc:
        return self._by_suffix.get(suffix.lower(), self._fallback)

    # --- 기본 fallback 구현 --------------------------------------------------------------

    def _default_fallback(self, path: Path) -> str:
        """
        기본 텍스트 추출기.

        - .txt/.md/.log 등 단순 텍스트 파일에 대해서는 UTF-8(+fallback)으로 읽기
        - 바이너리/이상한 포맷은 읽기 실패 시 빈 문자열 반환
        """
        try:
            # 가장 흔한 UTF-8부터 시도
            return path.read_text(encoding="utf-8", errors="ignore")
        except Exception as exc:  # noqa: BLE001
            logger.warning("Fallback extractor failed for %s: %s", path, exc)
            return ""


# === 메인 CorpusExtractor 구현 ============================================================

class CorpusExtractor:
    """
    스캔 결과(ScannedFile)에서 실제 코퍼스를 만드는 추출기.

    주요 책임:
    - allowed=1 인 파일만 대상으로 한다.
    - SmartFolderContext 범위 방어.
    - ExtractorRegistry를 사용해 확장자별 텍스트를 추출한다.
    - DocumentRecord 리스트 → pandas.DataFrame → Parquet/CSV 저장.
    """

    def __init__(
        self,
        smartfolder: SmartFolderContext,
        registry: Optional[ExtractorRegistry] = None,
        config: Optional[ExtractorConfig] = None,
    ) -> None:
        self.smartfolder = smartfolder
        self.registry = registry or ExtractorRegistry()
        self.config = config or ExtractorConfig(
            encoding_fallbacks=["utf-8", "cp949", "latin-1"]
        )

    # --- Public API ------------------------------------------------------------------

    def build_corpus_from_scanned(
        self,
        scanned_files: Iterable[ScannedFile],
    ) -> List[DocumentRecord]:
        """
        ScannedFile 리스트를 입력으로 받아 DocumentRecord 리스트를 생성.

        - allowed=1 인 파일만 대상
        - scope 밖인 파일은 방어적으로 제외
        """
        documents: List[DocumentRecord] = []

        for sf in scanned_files:
            if not sf.allowed:
                logger.debug("Skipping disallowed file: %s (reason=%s)", sf.path, sf.deny_reason)
                continue

            if not self.smartfolder.is_path_in_scope(sf.path):
                logger.warning("Scanned file is out of scope, skipping: %s", sf.path)
                continue

            try:
                doc = self._extract_document(sf)
            except Exception as exc:  # noqa: BLE001
                logger.warning("Failed to extract document from %s: %s", sf.path, exc)
                continue

            if not doc.text.strip():
                logger.debug("Empty text after extraction, skipping: %s", sf.path)
                continue

            documents.append(doc)

        logger.info("Corpus extraction finished. documents=%d", len(documents))
        return documents

    def build_corpus_from_csv(
        self,
        csv_path: Path,
    ) -> List[DocumentRecord]:
        """
        scanner.write_csv로 생성된 CSV에서 ScannedFile 목록을 복원한 뒤 코퍼스 생성.

        CSV 컬럼:
            path, size_bytes, modified_ts, content_hash, allowed, deny_reason
        """
        if not csv_path.exists():
            raise FileNotFoundError(f"scan csv not found: {csv_path}")

        df = pd.read_csv(csv_path)
        scanned_files: List[ScannedFile] = []

        for _, row in df.iterrows():
            sf = ScannedFile(
                path=Path(row["path"]),
                size_bytes=int(row["size_bytes"]),
                modified_ts=float(row["modified_ts"]),
                content_hash=str(row.get("content_hash", "") or "") or None,
                allowed=bool(int(row["allowed"])),
                deny_reason=str(row.get("deny_reason", "")) or None,
            )
            scanned_files.append(sf)

        return self.build_corpus_from_scanned(scanned_files)

    def corpus_to_dataframe(
        self,
        documents: Iterable[DocumentRecord],
    ) -> pd.DataFrame:
        """
        DocumentRecord 리스트를 pandas DataFrame으로 변환.
        """
        rows = [doc.to_row() for doc in documents]
        df = pd.DataFrame(rows)
        logger.debug("Converted corpus to DataFrame: shape=%s", df.shape)
        return df

    def save_corpus_parquet(
        self,
        df: pd.DataFrame,
        out_path: Path,
    ) -> None:
        """
        코퍼스를 Parquet 파일로 저장.

        - parent 디렉터리가 없으면 생성한다.
        """
        out_path.parent.mkdir(parents=True, exist_ok=True)
        logger.info("Saving corpus parquet: %s (rows=%d)", out_path, len(df))
        df.to_parquet(out_path, index=False)

    def save_corpus_csv(
        self,
        df: pd.DataFrame,
        out_path: Path,
    ) -> None:
        """
        코퍼스를 CSV 파일로 저장 (디버그/검수용).
        """
        out_path.parent.mkdir(parents=True, exist_ok=True)
        logger.info("Saving corpus csv: %s (rows=%d)", out_path, len(df))
        df.to_csv(out_path, index=False)

    # --- Internal helpers -------------------------------------------------------------

    def _extract_document(self, sf: ScannedFile) -> DocumentRecord:
        """
        단일 ScannedFile로부터 DocumentRecord 생성.

        - ExtractorRegistry를 이용해 텍스트 추출
        - 파일명/폴더명/확장자 기반 title/file_type/folder 설정
        - max_text_length 옵션 적용
        """
        path = sf.path
        suffix = path.suffix.lower()
        extractor = self.registry.get(suffix)

        raw_text = extractor(path)
        text = self._normalize_text(raw_text)

        max_len = self.config.max_text_length
        if max_len is not None and len(text) > max_len:
            logger.debug(
                "Truncating text for %s from length %d to %d",
                path,
                len(text),
                max_len,
            )
            text = text[:max_len]

        title = path.stem
        folder = str(path.parent)

        doc_id = self._build_doc_id(sf)

        metadata: Dict[str, Any] = {
            "size_bytes": sf.size_bytes,
            "modified_ts": sf.modified_ts,
            "content_hash": sf.content_hash,
            "suffix": suffix,
        }

        return DocumentRecord(
            doc_id=doc_id,
            path=path,
            text=text,
            title=title,
            file_type=suffix,
            folder=folder,
            metadata=metadata,
        )

    def _normalize_text(self, text: str) -> str:
        """
        텍스트 정규화 단계.

        - \r\n → \n
        - 이상하게 섞인 null 문자 제거 등
        - 향후 필요 시 추가 전처리 훅 연결 가능
        """
        if not text:
            return ""
        # 기본 정규화
        cleaned = text.replace("\r\n", "\n").replace("\r", "\n")
        cleaned = cleaned.replace("\x00", "")
        return cleaned

    def _build_doc_id(self, sf: ScannedFile) -> str:
        """
        문서 ID 생성 규칙.

        기본 구현:
        - content_hash가 있으면 그것을 사용
        - 없으면 path 기반 ID
        """
        if sf.content_hash:
            return f"hash:{sf.content_hash}"
        # content_hash가 없는 경우 path 기준으로 생성
        return f"path:{sf.path}"
```

# 2️⃣ 전체 스켈레톤: `embedder.py`

```python
"""
core.data_pipeline.embedder

역할:
- Extractor가 만든 코퍼스(DataFrame 또는 DocumentRecord 리스트)를 입력받아
  문서/청크 단위 임베딩을 생성한다.
- SentenceTransformer / ONNX / llama.cpp-compatible encoder 등 다양한 백엔드를
  추후 손쉽게 교체할 수 있는 인터페이스 계층을 제공한다.
- 생성된 임베딩은 Trainer/Retriever 단계에서 사용된다.

설계 핵심:
- Chunker (문서 → multiple chunks)
- EmbeddingBackend (다양한 모델 교체 가능)
- EmbeddingStoreWriter (parquet/csv/pickle 저장)
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, Iterable, List, Optional

import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)

# ======================================================================
# Chunker
# ======================================================================

@dataclass
class ChunkConfig:
    """
    문서 chunking 옵션.
    - chunk_size: 최대 토큰 또는 문자 단위
    - chunk_overlap: 슬라이딩 overlap 크기
    """
    chunk_size: int = 512
    chunk_overlap: int = 50
    strategy: str = "sliding"  # future: semantic_split 등 추가 가능


class Chunker:
    """
    텍스트를 model-ready chunk로 분할.

    기본 구현: 단순 sliding window 기반
    """

    def __init__(self, config: ChunkConfig) -> None:
        self.config = config

    def chunk_text(self, text: str) -> List[str]:
        if not text:
            return []

        size = self.config.chunk_size
        overlap = self.config.chunk_overlap

        chunks: List[str] = []
        start = 0
        N = len(text)

        while start < N:
            end = min(start + size, N)
            chunks.append(text[start:end])
            start += size - overlap

        return chunks


# ======================================================================
# Embedding Backend (모델 레이어)
# ======================================================================

class EmbeddingBackend:
    """
    임베딩 모델 백엔드 인터페이스.

    실제 구현체 예:
    - SentenceTransformerBackend
    - ONNXRuntimeBackend
    - LlamaCppEncoderBackend
    """

    def embed(self, texts: List[str]) -> np.ndarray:
        """
        입력 텍스트 리스트에 대한 (N, D) 벡터 배열을 반환.
        """
        raise NotImplementedError


# ---- 기본 SentenceTransformer backend ---------------------------------------------------

class SentenceTransformerBackend(EmbeddingBackend):
    """
    CPU-only 환경에서도 동작할 수 있는 기본 백엔드.

    model_name 예:
    - 'intfloat/multilingual-e5-small'
    - 'BAAI/bge-m3'
    """

    def __init__(self, model_name: str):
        try:
            from sentence_transformers import SentenceTransformer
        except ImportError as exc:  # noqa: BLE001
            raise RuntimeError("sentence-transformers 패키지가 필요합니다.") from exc

        self.model_name = model_name
        self.model = SentenceTransformer(model_name)

    def embed(self, texts: List[str]) -> np.ndarray:
        embeddings = self.model.encode(
            texts,
            show_progress_bar=False,
            convert_to_numpy=True,
            normalize_embeddings=True,
        )
        return embeddings


# ======================================================================
# EmbedderConfig
# ======================================================================

@dataclass
class EmbedderConfig:
    model_name: str = "intfloat/multilingual-e5-small"
    batch_size: int = 16
    # drift 검사/증분 재계산 등을 위한 hook
    ignore_empty_chunks: bool = True


# ======================================================================
# EmbeddingStore Writer
# ======================================================================

class EmbeddingStoreWriter:
    """
    생성된 임베딩과 메타데이터를 parquet/csv로 저장.
    retriever는 이 저장소를 불러와 벡터 검색을 수행한다.
    """

    def save_parquet(self, df: pd.DataFrame, out_path: Path) -> None:
        out_path.parent.mkdir(parents=True, exist_ok=True)
        logger.info("Saving embeddings parquet: %s (rows=%d)", out_path, len(df))
        df.to_parquet(out_path, index=False)

    def save_csv(self, df: pd.DataFrame, out_path: Path) -> None:
        out_path.parent.mkdir(parents=True, exist_ok=True)
        logger.info("Saving embeddings csv: %s (rows=%d)", out_path, len(df))
        df.to_csv(out_path, index=False)


# ======================================================================
# Main Embedder
# ======================================================================

class CorpusEmbedder:
    """
    Extractor → Embedder → Trainer/Retriever 전체 파이프라인에서
    '임베딩 생성 단계' 담당.

    입력:
        - corpus_df: DocumentRecord 기반 코퍼스 DataFrame (doc_id, text, ...)

    출력:
        - emb_df: chunk-level 기록 (chunk_id, doc_id, chunk_text, vector)
    """

    def __init__(
        self,
        backend: EmbeddingBackend,
        chunker: Chunker,
        config: Optional[EmbedderConfig] = None,
    ):
        self.backend = backend
        self.chunker = chunker
        self.config = config or EmbedderConfig()

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def build_embeddings(
        self,
        corpus_df: pd.DataFrame,
    ) -> pd.DataFrame:
        """
        corpus_df(text 컬럼 반드시 필요) → chunk-level 임베딩 DataFrame.
        """
        if "text" not in corpus_df.columns:
            raise ValueError("corpus_df must contain a 'text' column")

        rows: List[Dict[str, Any]] = []

        for idx, row in corpus_df.iterrows():
            doc_id = row["doc_id"]
            text = row["text"] or ""

            chunks = self.chunker.chunk_text(text)
            if not chunks and self.config.ignore_empty_chunks:
                continue

            # batch-size 수준으로 나눠 벡터 생성
            for batch in self._batch(chunks, self.config.batch_size):
                vectors = self.backend.embed(batch)
                for chunk, vec in zip(batch, vectors):
                    rows.append(
                        {
                            "doc_id": doc_id,
                            "chunk_text": chunk,
                            "embedding": vec,
                        }
                    )

        emb_df = pd.DataFrame(rows)
        logger.info("Embedding build finished. rows=%d", len(emb_df))
        return emb_df

    # ------------------------------------------------------------------

    def save_embeddings(
        self,
        emb_df: pd.DataFrame,
        out_path: Path,
        writer: Optional[EmbeddingStoreWriter] = None,
    ) -> None:
        writer = writer or EmbeddingStoreWriter()
        writer.save_parquet(emb_df, out_path)

    # ------------------------------------------------------------------

    def _batch(self, seq: List[Any], size: int) -> Iterable[List[Any]]:
        for i in range(0, len(seq), size):
            yield seq[i : i + size]
```

## 2️⃣ 전체 스켈레톤: `core/data_pipeline/trainer.py`

```python
"""
core.data_pipeline.trainer

역할:
- embedder 단계에서 생성된 chunk-level 임베딩을 기반으로
  토픽 모델 / 클러스터 / 품질 지표를 학습한다.
- 결과를 토픽 모델 아티팩트(topic_model.joblib) 및
  분석 리포트(JSON 등)로 저장해 retriever/RAG가 활용할 수 있게 한다.

설계 포인트:
- 모델 타입에 의존하지 않는 Trainer 인터페이스
- KMeans, HDBSCAN, NMF 등 다양한 백엔드로 교체 가능
- Smart Folder 컨텍스트에 맞는 최소 정보만 보존 (불필요한 개인정보/본문 재저장 금지)
"""

from __future__ import annotations

import json
import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, Optional, Tuple

import joblib
import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)


# ======================================================================
# Config
# ======================================================================

@dataclass
class TrainerConfig:
    """
    Trainer 동작 옵션.

    - n_topics: 토픽/클러스터 수 (KMeans 기준)
    - random_state: 재현성 확보용 seed
    - min_docs_per_topic: 너무 작은 클러스터는 '기타'로 묶기 위한 임계값
    """
    n_topics: int = 20
    random_state: int = 42
    min_docs_per_topic: int = 5


# ======================================================================
# Base Topic Model Wrapper
# ======================================================================

class TopicModelWrapper:
    """
    토픽/클러스터 모델 추상화 레이어.

    구현체 예:
    - KMeansTopicModel
    - NMFTopicModel
    - HDBSCANTopicModel
    """

    def fit(self, X: np.ndarray) -> None:  # X: (N, D)
        raise NotImplementedError

    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        각 샘플의 topic/cluster id를 반환.
        """
        raise NotImplementedError

    def save(self, path: Path) -> None:
        """
        모델 상태를 디스크에 저장.
        """
        raise NotImplementedError


class KMeansTopicModel(TopicModelWrapper):
    """
    가장 단순한 KMeans 기반 토픽 모델.

    - chunk-level 벡터에 대해 KMeans를 학습하고
    - 각 chunk에 cluster_id를 부여한다.
    """

    def __init__(self, n_clusters: int, random_state: int = 42) -> None:
        try:
            from sklearn.cluster import KMeans
        except ImportError as exc:  # noqa: BLE001
            raise RuntimeError("scikit-learn 패키지가 필요합니다.") from exc

        self._n_clusters = n_clusters
        self._random_state = random_state
        self._model = KMeans(
            n_clusters=n_clusters,
            random_state=random_state,
            n_init="auto",
        )

    def fit(self, X: np.ndarray) -> None:
        self._model.fit(X)

    def predict(self, X: np.ndarray) -> np.ndarray:
        return self._model.predict(X)

    def save(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        joblib.dump(
            {
                "model_type": "kmeans",
                "n_clusters": self._n_clusters,
                "random_state": self._random_state,
                "model": self._model,
            },
            path,
        )


# ======================================================================
# Trainer 본체
# ======================================================================

class CorpusTrainer:
    """
    Embedding → Topic Model 학습 → 메타데이터/리포트 생성 담당.

    입력:
        - emb_df: DataFrame
          - 필수 컬럼:
            - doc_id
            - embedding (np.ndarray 또는 list[float])

    출력:
        - topic_model.joblib (또는 지정 경로)
        - training_report.json (토픽 분포, 샘플 수, outlier 비율 등)
        - emb_df에 topic_id 컬럼을 추가한 DataFrame (옵션)
    """

    def __init__(
        self,
        config: Optional[TrainerConfig] = None,
        model: Optional[TopicModelWrapper] = None,
    ) -> None:
        self.config = config or TrainerConfig()
        self.model = model or KMeansTopicModel(
            n_clusters=self.config.n_topics,
            random_state=self.config.random_state,
        )

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def train(
        self,
        emb_df: pd.DataFrame,
    ) -> Tuple[pd.DataFrame, Dict[str, Any]]:
        """
        chunk-level 임베딩에 대해 토픽 모델을 학습하고
        topic_id를 부여한 DataFrame과 학습 리포트를 반환한다.
        """
        if "embedding" not in emb_df.columns:
            raise ValueError("emb_df must contain an 'embedding' column")

        # embedding 컬럼을 (N, D) np.ndarray로 변환
        X = self._to_matrix(emb_df["embedding"])
        if X.size == 0:
            raise ValueError("No embeddings found for training")

        logger.info(
            "Training topic model on embeddings. samples=%d, dim=%d",
            X.shape[0],
            X.shape[1],
        )

        self.model.fit(X)
        topic_ids = self.model.predict(X)

        # 결과 DataFrame에 topic_id 컬럼 추가
        emb_df = emb_df.copy()
        emb_df["topic_id"] = topic_ids

        # 학습 리포트 생성
        report = self._build_training_report(topic_ids)

        logger.info("Training finished. topics=%d", len(report.get("topic_counts", {})))
        return emb_df, report

    def save_artifacts(
        self,
        emb_df_with_topics: pd.DataFrame,
        report: Dict[str, Any],
        model_path: Path,
        report_path: Path,
    ) -> None:
        """
        - 토픽 모델을 model_path(joblib)에 저장
        - 학습 리포트를 JSON으로 저장
        - emb_df_with_topics는 상위 레이어에서 Parquet/CSV 등으로 저장
        """
        # 모델
        self.model.save(model_path)
        logger.info("Topic model saved: %s", model_path)

        # 리포트
        report_path.parent.mkdir(parents=True, exist_ok=True)
        with report_path.open("w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        logger.info("Training report saved: %s", report_path)

    # ------------------------------------------------------------------
    # Internal Helpers
    # ------------------------------------------------------------------

    def _to_matrix(self, series: pd.Series) -> np.ndarray:
        """
        embedding 컬럼(pd.Series of np.ndarray/list)을 (N, D) np.ndarray로 변환.
        """
        # 리스트/array 혼재 가능 → np.stack
        arr_list = []
        for v in series:
            # 이미 np.ndarray면 그대로, list면 np.array로
            if isinstance(v, np.ndarray):
                arr_list.append(v)
            else:
                arr_list.append(np.array(v, dtype=np.float32))

        if not arr_list:
            return np.empty((0, 0), dtype=np.float32)

        return np.stack(arr_list, axis=0)

    def _build_training_report(self, topic_ids: np.ndarray) -> Dict[str, Any]:
        """
        topic_id 분포 및 간단한 통계를 리포트로 반환.
        """
        unique, counts = np.unique(topic_ids, return_counts=True)
        topic_counts = {int(k): int(v) for k, v in zip(unique, counts)}

        total = int(topic_ids.shape[0])
        max_topic = max(topic_counts.values()) if topic_counts else 0
        min_topic = min(topic_counts.values()) if topic_counts else 0

        # 너무 작은 클러스터 비율 등
        small_threshold = self.config.min_docs_per_topic
        small_topics = {k: v for k, v in topic_counts.items() if v < small_threshold}

        report: Dict[str, Any] = {
            "total_chunks": total,
            "num_topics": len(topic_counts),
            "topic_counts": topic_counts,
            "max_topic_size": max_topic,
            "min_topic_size": min_topic,
            "small_topics": small_topics,
            "min_docs_per_topic": self.config.min_docs_per_topic,
        }

        return report
```

## 1️⃣ 역할 정의 – 새 `retriever`가 해야 하는 일

이 모듈은 한 줄로 요약하면:

> **“스마트 폴더 경계 안에서, 안전하게, 안정적으로, 인용 가능한 검색/RAG 결과를 돌려주는 엔진”**

구체적으로:

1. **입력**

   * `query: str`
   * `SearchContext`

     * `smart_folder_id` (or scope)
     * `policy` (allowed_types, redaction, 금지 패턴 등)
     * `top_k`, `lexical_weight`, `min_score` 등 옵션

2. **내부 처리**

   * 쿼리 전처리/정규화 (lower, stopword, 언어 감지 등 단순 단계)
   * 쿼리 임베딩 생성 (Embedder와 같은 모델 or 공유 인퍼런스)
   * **벡터 인덱스 + BM25 (optional)** 하이브리드 검색
   * 스마트 폴더/정책 필터 (scope 밖 문서/타입 제거)
   * 스코어 재정렬 (topic, recency, policy penalty 등 반영)
   * 결과를 **RAG/LLM이 바로 쓸 수 있는 구조**로 포맷팅

3. **출력**

   * `SearchResult` (리스트)

     * `doc_id`, `chunk_id`, `score`
     * `snippet`, `metadata` (path, title, created_at, tags, topic_id…)
     * `smart_folder_id`, `policy_tags`, `citations` 등

이 모듈이 잘 설계돼야, 위에 올라가는 **회의 에이전트 / 지식 에이전트 / HR 에이전트**가 전부 재사용 가능해진다.

## 3️⃣ 내부 컴포넌트 설계

### ① Index & Cache Layer

컴포넌트:

* `VectorIndexBackend`

  * 추상화: FAISS / sklearn NearestNeighbors / 단순 cosine search 등 무엇이든 교체 가능
  * 메서드:

    * `load(index_path, metadata_path)`
    * `search(query_vector, top_k, filter_mask=None) -> (indices, scores)`

* `LexicalIndexBackend` (optional)

  * whoosh/rapidfuzz/BM25 등 텍스트 기반 보조 검색
  * 메서드:

    * `search(query_text, top_k, filter_mask=None)`

* `IndexCache`

  * **락 없는(read-only) 캐시 컨셉**:

    * 인덱스는 “스냅샷 단위 파일”로 만들고
    * 프로세스 안에서는 read-only 객체로 올려놓음
    * 재빌드 시에는 새 스냅샷을 만들고 **핸들 교체**만 함
  * 메서드:

    * `get_index(smart_folder_id) -> LoadedIndex`
    * `invalidate(smart_folder_id)` (drift reembed 후)

### ② Smart Folder Guard + Policy Filter

* `SmartFolderGuard`

  * `smart_folders.json` + 정책 파일을 로드해
  * 인덱스와 결과 리스트에 대해 “이 scope에 포함되나?” 체크
  * 메서드:

    * `build_filter_mask(index_metadata, smart_folder_id) -> np.ndarray[bool]`
    * `apply_policy(results, policy) -> filtered_results, warnings`

정책 예:

* allowed_types: [`pdf`, `docx`, `md`]
* forbidden_patterns: 이메일/주민번호 등 정규식 → 스니펫 마스킹
* path_prefix: `/Users/me/Projects/foo/**`

### ③ Hybrid Scorer

* 입력:

  * `dense_scores` (vector similarity)
  * `lex_scores` (optional)
* 출력:

  * `hybrid_scores = (1 - w) * dense + w * lex`
  * 이후 topic/recency 기반 추가 보정도 hook으로 넣을 수 있음

### ④ Result Formatter

* `SearchResultFormatter`

  * chunk 텍스트 → snippet 생성

    * `max_snippet_length` 기준 자르기
    * 질의 단어 하이라이트 정보 포함 (필요 시 UI 쪽에서 쓰도록)
  * `SearchResultItem` 리스트 생성
  * 정책에 의해 마스킹된 부분에는 `[REDACTED]` 토큰 등 삽입

## 4️⃣ `core/search/retriever.py` 스켈레톤 (코드 구조만)

> 실제 구현 대신 **뼈대 + 인터페이스** 위주로 적는다.
> (네 팀이 이 틀에 맞춰 현재 `retriever.py`/`lnp_chat.py`를 옮겨 타면 됨.)

```python
"""
core.search.retriever

역할:
- Smart Folder 컨텍스트 안에서 안전하게 dense/lexical 하이브리드 검색을 수행.
- RAG/대화 에이전트가 바로 사용할 수 있는 SearchResult 구조를 반환.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, List, Optional

import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)


# ======================================================================
# Public Dataclasses
# ======================================================================

@dataclass
class SearchConfig:
    top_k: int = 10
    lexical_weight: float = 0.35
    min_score: float = 0.0
    max_snippet_length: int = 300
    # 향후: recency_weight, topic_boost 등 추가 가능


@dataclass
class SearchContext:
    smart_folder_id: str
    policy: Dict[str, Any]
    user_id: Optional[str] = None
    session_id: Optional[str] = None
    config: Optional[SearchConfig] = None


@dataclass
class SearchResultItem:
    doc_id: str
    chunk_id: int
    score: float
    snippet: str
    source_path: str
    topic_id: Optional[int] = None
    metadata: Dict[str, Any] = None
    policy_tags: List[str] = None


@dataclass
class SearchResult:
    query: str
    items: List[SearchResultItem]
    smart_folder_id: str
    policy_warnings: List[str]


# ======================================================================
# Index Backends (Interfaces)
# ======================================================================

class VectorIndexBackend:
    """
    벡터 인덱스 추상화.

    구현 예:
    - FAISSIndexBackend
    - NumpyCosineBackend
    """

    def search(
        self,
        query_vector: np.ndarray,
        top_k: int,
        filter_mask: Optional[np.ndarray] = None,
    ) -> (np.ndarray, np.ndarray):
        """
        Returns:
            indices: (top_k,)
            scores: (top_k,)
        """
        raise NotImplementedError


class LexicalIndexBackend:
    """
    BM25 / 후보 문서 텍스트 기반 보조 검색용 백엔드.
    """

    def search(
        self,
        query_text: str,
        top_k: int,
        filter_mask: Optional[np.ndarray] = None,
    ) -> (np.ndarray, np.ndarray):
        raise NotImplementedError


# ======================================================================
# Smart Folder Guard
# ======================================================================

class SmartFolderGuard:
    """
    smart_folders.json + policy를 기반으로
    인덱스/결과에 대한 scope & policy 필터링을 수행.
    """

    def __init__(self, smart_folders_path: Path):
        self._smart_folders_path = smart_folders_path
        self._meta = self._load_smart_folders()

    def _load_smart_folders(self) -> Dict[str, Any]:
        # NOTE: 실제 구현은 `core/data_pipeline/policies/loader/` + `core/data_pipeline/policies/engine.py`에서
        # jsonschema 기반 검증까지 포함해 수행한다. (여기 코드는 개념 스켈레톤)
        return {}

    def build_filter_mask(
        self,
        index_metadata: pd.DataFrame,
        smart_folder_id: str,
        policy: Dict[str, Any],
    ) -> np.ndarray:
        """
        index_metadata(row별 path, file_type 등) 기준으로
        현재 smart_folder_id + policy에 맞는 bool mask 생성.
        """
        # NOTE: 현 구현에서는 "인덱스 결과에 마스크를 씌우는 방식" 대신,
        # scan 단계에서 `allowed`를 계산하고 이후 단계에서 `allowed=0`을 fail-closed로 제외한다.
        # (PolicyEngine: `core/data_pipeline/policies/engine.py`)
        return np.ones(len(index_metadata), dtype=bool)

    def apply_policy_to_snippet(
        self,
        snippet: str,
        policy: Dict[str, Any],
    ) -> (str, List[str]):
        """
        snippet에 민감 정보 마스킹 정책 적용.
        Returns:
            masked_snippet, policy_tags
        """
        # NOTE: 마스킹은 에이전트별 파이프라인(예: `core/agents/meeting/pii.py`)과
        # 정책 토글(`PolicyEngine.pii_mask_enabled_for_path`)에서 수행한다.
        return snippet, []


# ======================================================================
# Retriever 본체
# ======================================================================

class Retriever:
    """
    하이브리드 검색 엔진.

    구성 요소:
    - vector_backend: dense search
    - lexical_backend: optional lexical search
    - guard: smart folder + policy 필터
    - index_metadata: doc_id, chunk_id, path, topic_id 등 메타데이터 DataFrame
    """

    def __init__(
        self,
        vector_backend: VectorIndexBackend,
        lexical_backend: Optional[LexicalIndexBackend],
        index_metadata: pd.DataFrame,
        guard: SmartFolderGuard,
        default_config: Optional[SearchConfig] = None,
    ) -> None:
        self.vector_backend = vector_backend
        self.lexical_backend = lexical_backend
        self.index_metadata = index_metadata
        self.guard = guard
        self.default_config = default_config or SearchConfig()

    # ------------------------------------------------------------------

    def search(
        self,
        query: str,
        query_vector: np.ndarray,
        context: SearchContext,
    ) -> SearchResult:
        """
        query_text + query_vector를 받아 smart_folder 범위 내에서 검색을 수행.
        """
        cfg = context.config or self.default_config

        # 1) smart folder + policy 기반 mask 생성
        mask = self.guard.build_filter_mask(
            index_metadata=self.index_metadata,
            smart_folder_id=context.smart_folder_id,
            policy=context.policy,
        )

        # 2) dense 검색
        dense_idx, dense_scores = self.vector_backend.search(
            query_vector=query_vector,
            top_k=cfg.top_k,
            filter_mask=mask,
        )

        # 3) lexical 검색 (있을 경우)
        if self.lexical_backend is not None and cfg.lexical_weight > 0:
            lex_idx, lex_scores = self.lexical_backend.search(
                query_text=query,
                top_k=cfg.top_k,
                filter_mask=mask,
            )
        else:
            lex_idx, lex_scores = None, None

        # 4) 하이브리드 스코어링 + 정렬
        final_indices, final_scores = self._merge_scores(
            dense_idx,
            dense_scores,
            lex_idx,
            lex_scores,
            cfg.lexical_weight,
        )

        # 5) 결과 포맷팅 + 정책 마스킹
        items, policy_warnings = self._build_results(
            final_indices,
            final_scores,
            cfg,
            context,
        )

        return SearchResult(
            query=query,
            items=items,
            smart_folder_id=context.smart_folder_id,
            policy_warnings=policy_warnings,
        )

    # ------------------------------------------------------------------

    def _merge_scores(
        self,
        dense_idx: np.ndarray,
        dense_scores: np.ndarray,
        lex_idx: Optional[np.ndarray],
        lex_scores: Optional[np.ndarray],
        lexical_weight: float,
    ) -> (np.ndarray, np.ndarray):
        """
        dense + lexical를 하나의 score space로 병합.
        간단 버전: 같은 index 기준 가중 평균.
        """
        # NOTE: 현 구현은 `core/search/retriever.py`에서 dense(semantic) + BM25 점수를
        # 정규화 후 가중합(hybrid_score)으로 결합한다.
        return dense_idx, dense_scores

    # ------------------------------------------------------------------

    def _build_results(
        self,
        indices: np.ndarray,
        scores: np.ndarray,
        cfg: SearchConfig,
        context: SearchContext,
    ) -> (List[SearchResultItem], List[str]):
        """
        index_metadata를 참조해 SearchResultItem 리스트를 생성.
        """
        items: List[SearchResultItem] = []
        policy_warnings: List[str] = []

        for rank, (idx, score) in enumerate(zip(indices, scores)):
            if score < cfg.min_score:
                continue

            meta = self.index_metadata.iloc[int(idx)]
            raw_text = meta.get("chunk_text", "")
            topic_id = meta.get("topic_id")
            source_path = meta.get("path", "")
            doc_id = meta.get("doc_id", "")

            snippet = raw_text[: cfg.max_snippet_length]
            masked_snippet, policy_tags = self.guard.apply_policy_to_snippet(
                snippet,
                context.policy,
            )

            items.append(
                SearchResultItem(
                    doc_id=doc_id,
                    chunk_id=int(meta.get("chunk_id", rank)),
                    score=float(score),
                    snippet=masked_snippet,
                    source_path=source_path,
                    topic_id=int(topic_id) if topic_id is not None else None,
                    metadata={
                        "rank": rank,
                        "file_type": meta.get("file_type"),
                        "created_at": meta.get("created_at"),
                    },
                    policy_tags=policy_tags,
                )
            )

        # NOTE: 현 구현은 deny된 파일을 scan 단계에서 제외하고(`allowed=0`) 이후 단계를 진행한다.
        # 경고/요약은 필요 시 런타임 로그(`artifacts/logs/`)로 남기도록 확장 가능.
        return items, policy_warnings
```

  * FAISS/NP-cosine, BM25/none 같은 선택을 **교체 비용 최소화**로 엮을 수 있음.

### 🔹 UX 관점 (API UX)

* 상위 레벨에서 이 모듈을 쓰는 사용자는 딱 이렇게 보면 됨:

  ```python
  retriever.search(
      query="어제 회의에서 액션 아이템 뭐였지?",
      query_vector=<임베딩>,
      context=SearchContext(
          smart_folder_id="meeting_folder",
          policy=...,
      ),
  )
  ```

* 반환값에는 항상:

  * `snippet`, `source_path`, `topic_id`, `policy_tags`, `score` 가 포함

* 이걸 바로:

  * CLI 로그, Electron UI, 향후 웹 UI에 붙일 수 있음.

### 🔹 ML 관점

* Chunking / Embedding / Topic 모델을 이미 분리했으니,
* **검색 품질 개선 실험**은 여기서:

  * lexical_weight 튜닝
  * topic_boost (예: 사용자 도메인별 우선순위)
  * recency_boost (created_at 기반)
  * query expansion (간단 rule 기반)
* 단, 이 모듈 자체는 “순수 검색 로직”으로 유지하고,
  **LLM이 개입하는 RAG orchestrator**는 `core/conversation/lnp_chat.py`로 두는 편이 구조상 깔끔함.

여기까지가 **새 파이프라인 기준 `core/search/retriever.py` 전체 뼈대**야.
이제 이 위에 **대화/RAG 레이어 (lnp_chat 리팩터링)**를 얹어야 완성이다.

준비됐으면 다시 **“다음”**이라고 해.
다음 턴에서는:

> `core/conversation/lnp_chat.py`를
>
> * Smart Folder 컨텍스트
> * 새 retriever
> * Meeting/Knowledge Agent 호출 흐름
>   에 맞게 **엔드투엔드 오케스트레이션 스켈레톤**으로 재정의해줄게.

오케이, 이제 **대화/RAG 오케스트레이터 – `core/conversation/lnp_chat.py` 턴**으로 간다.
여기가 **“사용자 입력 → 검색/RAG → LLM 호출 → 응답” 전체 흐름의 허리**다.

## 1️⃣ 이 레이어의 역할 정리

한 줄로 요약하면:

> **사용자 질문 + Smart Folder 컨텍스트를 받아서, retriever를 돌리고, LLM 프롬프트를 구성해서, 최종 답변/인용/액션을 만들어내는 엔진**

구체적으로는:

1. **입력**

   * 자연어 질문 `user_input: str`
   * `ChatContext`

     * `smart_folder_id`
     * `agent_type` (meeting / knowledge / generic)
     * `policy` (스마트 폴더 정책)
     * `llm_backend`/`model_name` 등

2. **내부 단계**

   * 히스토리 로드 (이전 turn들)
   * 질의 분류:

     * 단순 Q&A?
     * “회의 요약/액션” 같은 **Meeting Agent 호출 필요?**
     * 일반 지식 검색?
   * `Retriever` 호출해서 RAG 컨텍스트(관련 문서 스니펫) 가져오기
   * 에이전트별 Prompt Template/Role 적용
   * LLM 호출 (로컬 무료 모델: llama.cpp/ollama/bitnet 등)
   * 응답 파싱:

     * 답변 텍스트
     * 액션 아이템 리스트 (회의 에이전트일 때)
     * 인용 문서 정보 (citations)

3. **출력**

   * `ChatResponse`

     * `answer_text`
     * `citations` (doc_id, path, snippet)
     * `actions` (optional; meeting)
     * `raw_model_output` (디버깅/로그용)

## 4️⃣ `core/conversation/lnp_chat.py` 스켈레톤

> 코드 “골격”만. 실제 구현은 개발팀이 이 틀에 맞춰 채우면 된다.

```python
"""
core.conversation.lnp_chat

역할:
- Smart Folder 컨텍스트와 Agent 타입을 고려하여
  검색(RAG) + LLM 호출을 오케스트레이션.
- Meeting/Knowledge 에이전트가 공통으로 사용하는 대화 엔진.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional

import numpy as np

from core.search.retriever import (
    Retriever,
    SearchContext,
    SearchConfig,
    SearchResult,
)
# from core.agents.meeting import MeetingAgent  # (향후)
# from core.agents.knowledge import KnowledgeAgent  # (향후)

logger = logging.getLogger(__name__)


# ======================================================================
# Config / Context / DTO
# ======================================================================

@dataclass
class ChatConfig:
    """
    LNP Chat 전체 동작 옵션.
    """
    max_history_turns: int = 10
    max_context_snippets: int = 5
    enable_rag: bool = True
    # LLM 옵션
    temperature: float = 0.2
    max_tokens: int = 1024
    model_name: str = "local-llm"
    # 검색 옵션 override
    search_config: SearchConfig = field(default_factory=SearchConfig)


@dataclass
class ChatContext:
    """
    한 번의 대화 세션을 설명하는 컨텍스트.
    """
    smart_folder_id: str
    agent_type: str  # "meeting" | "knowledge" | "generic"
    policy: Dict[str, Any]
    user_id: Optional[str] = None
    session_id: Optional[str] = None
    config: Optional[ChatConfig] = None


@dataclass
class ChatMessage:
    role: str  # "user" | "assistant" | "system"
    content: str


@dataclass
class ChatResponse:
    """
    상위 UI/CLI에 반환할 응답 구조.
    """
    answer_text: str
    citations: List[Dict[str, Any]]
    actions: List[Dict[str, Any]]
    raw_model_output: Optional[str] = None
    debug_info: Optional[Dict[str, Any]] = None


# ======================================================================
# LLM Backend (추상화)
# ======================================================================

class LLMBackend:
    """
    로컬/원격 상관없이 LLM 호출을 위한 추상 인터페이스.
    """

    def generate(self, prompt: str, **kwargs: Any) -> str:
        raise NotImplementedError


# ======================================================================
# RAG Orchestrator
# ======================================================================

class RAGOrchestrator:
    """
    Retriever를 감싸서:
    - 쿼리 임베딩
    - 검색 실행
    - 프롬프트에 들어갈 컨텍스트 포맷팅
    를 담당.
    """

    def __init__(self, retriever: Retriever, embed_fn) -> None:
        """
        embed_fn: query(str) -> np.ndarray
        """
        self.retriever = retriever
        self.embed_fn = embed_fn

    def retrieve(
        self,
        query: str,
        chat_ctx: ChatContext,
    ) -> SearchResult:
        cfg = chat_ctx.config.search_config if chat_ctx.config else SearchConfig()

        query_vec = self.embed_fn(query)

        search_ctx = SearchContext(
            smart_folder_id=chat_ctx.smart_folder_id,
            policy=chat_ctx.policy,
            user_id=chat_ctx.user_id,
            session_id=chat_ctx.session_id,
            config=cfg,
        )

        result = self.retriever.search(
            query=query,
            query_vector=query_vec,
            context=search_ctx,
        )
        return result

    def build_context_block(
        self,
        search_result: SearchResult,
        max_snippets: int,
    ) -> str:
        """
        검색 결과를 프롬프트에 삽입할 수 있는 컨텍스트 문자열로 변환.
        """
        items = search_result.items[:max_snippets]
        lines: List[str] = []

        for i, item in enumerate(items):
            lines.append(
                f"[DOC {i+1}] score={item.score:.3f} path={item.source_path}\n"
                f"{item.snippet}\n"
            )

        return "\n\n".join(lines)


# ======================================================================
# Prompt Builder
# ======================================================================

class PromptBuilder:
    """
    Agent 타입에 따라 다른 프롬프트 템플릿을 제공.
    """

    def build_prompt(
        self,
        agent_type: str,
        user_input: str,
        history: List[ChatMessage],
        context_block: str,
    ) -> str:
        if agent_type == "meeting":
            return self._build_meeting_prompt(user_input, history, context_block)
        if agent_type == "knowledge":
            return self._build_knowledge_prompt(user_input, history, context_block)
        return self._build_generic_prompt(user_input, history, context_block)

    # ------------------------------------------------------------------

    def _build_meeting_prompt(
        self,
        user_input: str,
        history: List[ChatMessage],
        context_block: str,
    ) -> str:
        # TODO: 실제 템플릿 정의
        return f"""
You are a meeting assistant. Summarize and extract action items.

# Context
{context_block}

# User request
{user_input}
        """.strip()

    def _build_knowledge_prompt(
        self,
        user_input: str,
        history: List[ChatMessage],
        context_block: str,
    ) -> str:
        return f"""
You are a knowledge base assistant. Answer using ONLY the given context.

# Context
{context_block}

# Conversation History
{self._format_history(history)}

# User request
{user_input}
        """.strip()

    def _build_generic_prompt(
        self,
        user_input: str,
        history: List[ChatMessage],
        context_block: str,
    ) -> str:
        return f"""
You are an AI assistant helping the user with their documents.

# Context
{context_block}

# Conversation History
{self._format_history(history)}

# User request
{user_input}
        """.strip()

    # ------------------------------------------------------------------

    def _format_history(self, history: List[ChatMessage]) -> str:
        lines = []
        for msg in history:
            lines.append(f"{msg.role.upper()}: {msg.content}")
        return "\n".join(lines)


# ======================================================================
# Conversation Store (간단 in-memory 버전)
# ======================================================================

class ConversationStore:
    """
    session_id 기준으로 히스토리를 관리.
    초기 버전은 in-memory, 나중에 파일/DB로 교체.
    """

    def __init__(self) -> None:
        self._store: Dict[str, List[ChatMessage]] = {}

    def get_history(self, session_id: str, limit: int) -> List[ChatMessage]:
        history = self._store.get(session_id, [])
        return history[-limit:]

    def append(
        self,
        session_id: str,
        user_msg: ChatMessage,
        assistant_msg: ChatMessage,
    ) -> None:
        history = self._store.setdefault(session_id, [])
        history.append(user_msg)
        history.append(assistant_msg)


# ======================================================================
# LNPChat – 메인 오케스트레이터
# ======================================================================

class LNPChat:
    """
    전체 대화/RAG 오케스트레이터.

    사용 예:
        chat = LNPChat(...)
        resp = chat.handle_turn("어제 회의 액션 정리해줘", chat_ctx)
    """

    def __init__(
        self,
        rag_orchestrator: RAGOrchestrator,
        llm_backend: LLMBackend,
        prompt_builder: PromptBuilder,
        conv_store: ConversationStore,
        default_config: Optional[ChatConfig] = None,
    ) -> None:
        self.rag = rag_orchestrator
        self.llm = llm_backend
        self.prompt_builder = prompt_builder
        self.store = conv_store
        self.default_config = default_config or ChatConfig()

    # ------------------------------------------------------------------

    def handle_turn(
        self,
        user_input: str,
        chat_ctx: ChatContext,
    ) -> ChatResponse:
        cfg = chat_ctx.config or self.default_config
        session_id = chat_ctx.session_id or "default"

        # 1) 히스토리 로드
        history = self.store.get_history(session_id, cfg.max_history_turns)

        # 2) (옵션) RAG 검색 실행
        context_block = ""
        search_result: Optional[SearchResult] = None
        if cfg.enable_rag:
            search_result = self.rag.retrieve(user_input, chat_ctx)
            context_block = self.rag.build_context_block(
                search_result,
                max_snippets=cfg.max_context_snippets,
            )

        # 3) 프롬프트 생성
        prompt = self.prompt_builder.build_prompt(
            agent_type=chat_ctx.agent_type,
            user_input=user_input,
            history=history,
            context_block=context_block,
        )

        # 4) LLM 호출
        raw_output = self.llm.generate(
            prompt,
            temperature=cfg.temperature,
            max_tokens=cfg.max_tokens,
        )

        # 5) 응답 파싱 (액션/인용 등)
        answer_text, actions = self._parse_model_output(
            raw_output,
            agent_type=chat_ctx.agent_type,
        )

        # 6) citations 구성 (검색 결과 기반)
        citations = self._build_citations(search_result)

        # 7) 히스토리 저장
        user_msg = ChatMessage(role="user", content=user_input)
        asst_msg = ChatMessage(role="assistant", content=answer_text)
        self.store.append(session_id, user_msg, asst_msg)

        return ChatResponse(
            answer_text=answer_text,
            citations=citations,
            actions=actions,
            raw_model_output=raw_output,
            debug_info={
                "context_block_preview": context_block[:500],
            },
        )

    # ------------------------------------------------------------------

    def _parse_model_output(
        self,
        raw_output: str,
        agent_type: str,
    ) -> (str, List[Dict[str, Any]]):
        """
        모델 출력에서:
        - 최종 답변 텍스트
        - 액션 아이템 (meeting일 때)
        을 추출.
        초기 버전은 단순 pass-through, 나중에 JSON 파싱 등 추가.
        """
        actions: List[Dict[str, Any]] = []

        if agent_type == "meeting":
            # NOTE: 현 구현은 `core/agents/meeting/pipeline.py::_extract_action_items()`에서
            # action items를 구조화해 생성한다.
            pass

        return raw_output.strip(), actions

    def _build_citations(
        self,
        search_result: Optional[SearchResult],
    ) -> List[Dict[str, Any]]:
        if search_result is None:
            return []

        cites: List[Dict[str, Any]] = []
        for i, item in enumerate(search_result.items):
            cites.append(
                {
                    "rank": i,
                    "doc_id": item.doc_id,
                    "path": item.source_path,
                    "score": item.score,
                    "topic_id": item.topic_id,
                }
            )
        return cites
```
